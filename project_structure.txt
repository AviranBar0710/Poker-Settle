Poker Settle — Project Structure

│   ├── page.tsx                          # Home / dashboard (session list)

│   ├── profile/page.tsx                  # User profile settings
│   ├── session/
│   │   ├── [id]/page.tsx                # Live session detail page
│   │   └── join/[token]/page.tsx        # Join session via invite link
│   ├── sessions/page.tsx                 # All sessions list
│   ├── stats/

│   ├── session/

│   │   ├── EmptyState.tsx                # No-players empty state

│       └── table.tsx

│       └── utils.ts                      # Helpers
